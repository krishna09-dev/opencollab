## Overview

The ML model in OpenCollab is a **content-based recommendation system** that matches beginner developers to suitable GitHub issues based on:
- The **text content** of issues (title + description + labels)
- The **user's preferences** (programming language, difficulty level, topics of interest)

The model uses **TF-IDF (Term Frequency-Inverse Document Frequency)** vectorization combined with **Cosine Similarity** to rank issues by relevance to a user profile.

### Goal
> Given a user profile with preferred languages and skill level, return a ranked list of beginner-friendly GitHub issues most relevant to that user.

### ML Pipeline Overview
```
Raw Data → Preprocessing → TF-IDF Vectorization → Cosine Similarity → Ranked Recommendations
```

## Phase 3 – Data Preprocessing

### 3.1 Goals
- Remove noise and irrelevant characters
- Combine relevant text fields into one `text` field for vectorization
- Encode categorical fields numerically

### 3.2 Preprocessing Script

```python
# preprocess.py
import pandas as pd
import re
import nltk
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer

nltk.download("stopwords")
stop_words = set(stopwords.words("english"))
stemmer = PorterStemmer()

def clean_text(text: str) -> str:
    """Lowercase, remove special chars, remove stopwords, stem."""
    if not isinstance(text, str):
        return ""
    text = text.lower()
    text = re.sub(r"[^a-z0-9\s]", " ", text)        # Remove special chars
    text = re.sub(r"\s+", " ", text).strip()          # Collapse whitespace
    tokens = text.split()
    tokens = [t for t in tokens if t not in stop_words]  # Remove stopwords
    tokens = [stemmer.stem(t) for t in tokens]            # Stem
    return " ".join(tokens)

def preprocess(df: pd.DataFrame) -> pd.DataFrame:
    # Combine fields into one text blob for vectorization
    df["combined_text"] = (
        df["issue_title"].fillna("") + " " +
        df["issue_body"].fillna("") + " " +
        df["labels"].fillna("") + " " +
        df["language"].fillna("") + " " +
        df["topics"].fillna("")
    )

    df["combined_text_clean"] = df["combined_text"].apply(clean_text)

    # Encode difficulty as numeric (for filtering)
    difficulty_map = {"beginner": 1, "intermediate": 2, "advanced": 3}
    df["difficulty_score"] = df["difficulty"].map(difficulty_map)

    # Filter: only open issues
    df = df[df["is_open"] == True].reset_index(drop=True)

    return df

if __name__ == "__main__":
    df = pd.read_csv("data/raw_issues.csv")
    df = preprocess(df)
    df.to_csv("data/processed_issues.csv", index=False)
    print(f"Preprocessed {len(df)} open issues")
    print(df[["issue_title", "combined_text_clean", "difficulty"]].head(3))
```

## Phase 4 – Feature Engineering

### 4.1 User Profile Vector

A user profile is constructed from their settings and represented as a text query:

```python
def build_user_query(user_profile: dict) -> str:
    """
    Convert user preferences into a text query for similarity matching.
    
    Example user_profile:
    {
        "languages": ["Python", "JavaScript"],
        "difficulty": "beginner",
        "topics": ["web", "api"],
        "keywords": ["documentation", "test", "bug fix"]
    }
    """
    parts = []
    parts.extend(user_profile.get("languages", []))
    parts.append(user_profile.get("difficulty", "beginner"))
    parts.extend(user_profile.get("topics", []))
    parts.extend(user_profile.get("keywords", []))

    # Repeat preferred language to give it more weight
    preferred_lang = user_profile.get("languages", [""])[0]
    parts.extend([preferred_lang] * 3)  # Boost language weight

    return " ".join(parts).lower()
```

## Phase 5 – Model Building (TF-IDF)

### 5.1 What is TF-IDF?

TF-IDF converts text into numerical vectors by:
- **TF (Term Frequency):** How often a word appears in a document
- **IDF (Inverse Document Frequency):** How rare a word is across all documents
- Words that appear frequently in one document but rarely elsewhere get **high scores** (they are distinctive)

### 5.2 Model Training Script

```python
# model.py
import pandas as pd
import numpy as np
import pickle
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def train_tfidf_model(df: pd.DataFrame):
    """
    Fit TF-IDF vectorizer on the cleaned issue text corpus.
    Returns the fitted vectorizer and the issue matrix.
    """
    vectorizer = TfidfVectorizer(
        max_features=5000,       # Top 5000 most important words
        ngram_range=(1, 2),      # Unigrams and bigrams
        min_df=2,                # Ignore words appearing in fewer than 2 docs
        max_df=0.85,             # Ignore words appearing in more than 85% of docs
        sublinear_tf=True        # Apply log normalization to TF
    )

    tfidf_matrix = vectorizer.fit_transform(df["combined_text_clean"])
    print(f"TF-IDF Matrix Shape: {tfidf_matrix.shape}")
    # e.g., (425, 5000) → 425 issues × 5000 features

    return vectorizer, tfidf_matrix

def save_model(vectorizer, tfidf_matrix, df):
    """Save all model artifacts for use in FastAPI."""
    with open("model/vectorizer.pkl", "wb") as f:
        pickle.dump(vectorizer, f)

    np.save("model/tfidf_matrix.npy", tfidf_matrix.toarray())
    df.to_csv("model/issues_index.csv", index=False)
    print("Model artifacts saved.")

if __name__ == "__main__":
    df = pd.read_csv("data/processed_issues.csv")
    vectorizer, tfidf_matrix = train_tfidf_model(df)
    save_model(vectorizer, tfidf_matrix, df)
```

## Phase 6 – Recommendation Logic

### 6.1 How Recommendations Work

```
User Profile (text query)
        ↓
TF-IDF Vectorize the query (same vectorizer)
        ↓
Cosine Similarity vs. all issue vectors
        ↓
Filter by difficulty & language
        ↓
Return Top-N ranked issues
```

### 6.2 Recommendation Script

```python
# recommend.py
import pandas as pd
import numpy as np
import pickle
from sklearn.metrics.pairwise import cosine_similarity
from preprocess import clean_text
from feature_engineering import build_user_query

def load_model():
    with open("model/vectorizer.pkl", "rb") as f:
        vectorizer = pickle.load(f)
    tfidf_matrix = np.load("model/tfidf_matrix.npy")
    df = pd.read_csv("model/issues_index.csv")
    return vectorizer, tfidf_matrix, df

def recommend_issues(user_profile: dict, top_n: int = 10) -> pd.DataFrame:
    vectorizer, tfidf_matrix, df = load_model()

    # Step 1: Build and vectorize user query
    query = build_user_query(user_profile)
    query_clean = clean_text(query)
    query_vec = vectorizer.transform([query_clean])

    # Step 2: Compute cosine similarity
    similarities = cosine_similarity(query_vec, tfidf_matrix).flatten()
    df["similarity_score"] = similarities

    # Step 3: Filter by difficulty (only show issues at or below user level)
    difficulty_map = {"beginner": 1, "intermediate": 2, "advanced": 3}
    user_level = difficulty_map.get(user_profile.get("difficulty", "beginner"), 1)
    df_filtered = df[df["difficulty_score"] <= user_level].copy()

    # Step 4: Filter by preferred language (optional but boosted)
    preferred_langs = [l.lower() for l in user_profile.get("languages", [])]
    if preferred_langs:
        lang_mask = df_filtered["language"].str.lower().isin(preferred_langs)
        df_lang = df_filtered[lang_mask]
        df_other = df_filtered[~lang_mask]
        # Prioritize language matches, then append others
        df_filtered = pd.concat([df_lang, df_other])

    # Step 5: Sort by similarity score and return top N
    result = df_filtered.sort_values("similarity_score", ascending=False).head(top_n)

    return result[["issue_id", "repo_name", "issue_title", "language",
                    "difficulty", "labels", "similarity_score"]]


# --- Example Usage ---
if __name__ == "__main__":
    user = {
        "languages": ["Python"],
        "difficulty": "beginner",
        "topics": ["web", "api"],
        "keywords": ["documentation", "test"]
    }
    recommendations = recommend_issues(user, top_n=5)
    print(recommendations.to_string(index=False))
```

## Phase 7 – Model Evaluation

### 7.1 Evaluation Strategy

Since this is an unsupervised recommendation system (no explicit user ratings), evaluation is done using:

1. **Manual Relevance Scoring** – human evaluators rate recommendations
2. **Precision@K** – how many of the top-K results are actually relevant
3. **Diversity Score** – are recommendations varied (different repos/languages)?
4. **Coverage** – does the model recommend items from across the dataset?
5. **Cold Start Test** – does the model handle brand new users with minimal preferences?

### 7.3 Precision@K

```python
def precision_at_k(recommendations: pd.DataFrame, expected_lang: str,
                    expected_difficulty: str, k: int = 5) -> float:
    """
    Measure what fraction of top-K results match expected language and difficulty.
    """
    top_k = recommendations.head(k)
    relevant = 0

    for _, row in top_k.iterrows():
        lang_match = (expected_lang is None) or (row["language"] == expected_lang)
        diff_match = (row["difficulty"] == expected_difficulty)
        if lang_match and diff_match:
            relevant += 1

    return relevant / k

def run_evaluation():
    results = []
    for test in TEST_USERS:
        recs = recommend_issues(test["profile"], top_n=10)
        p_at_5 = precision_at_k(recs, test["expected_lang"],
                                  test["expected_difficulty"], k=5)
        p_at_10 = precision_at_k(recs, test["expected_lang"],
                                   test["expected_difficulty"], k=10)
        results.append({
            "user_id": test["id"],
            "precision@5": round(p_at_5, 2),
            "precision@10": round(p_at_10, 2),
            "top_recommendation": recs.iloc[0]["issue_title"] if len(recs) > 0 else "None"
        })

    results_df = pd.DataFrame(results)
    print(results_df.to_string(index=False))
    results_df.to_csv("evaluation/eval_results.csv", index=False)

if __name__ == "__main__":
    run_evaluation()

def diversity_score(recommendations: pd.DataFrame) -> float:
    """Fraction of unique repos in top-10 recommendations."""
    unique_repos = recommendations["repo_name"].nunique()
    return unique_repos / len(recommendations)

def coverage_score(recommendations: pd.DataFrame, total_issues: int) -> float:
    """Fraction of total dataset represented in recommendations."""
    return len(recommendations) / total_issues
```

## Phase 8 – Fallback Strategy

If TF-IDF similarity scores are all below a threshold (poor match), the system falls back to **label-based filtering**.

```python
def fallback_recommend(user_profile: dict, df: pd.DataFrame,
                        top_n: int = 10) -> pd.DataFrame:
    """
    Simple label-based fallback when TF-IDF gives poor scores.
    Filters by difficulty and language only.
    """
    print("Using fallback: label-based filtering")

    # Filter by difficulty
    difficulty = user_profile.get("difficulty", "beginner")
    df_filtered = df[df["difficulty"] == difficulty].copy()

    # Filter by language if specified
    preferred_langs = [l.lower() for l in user_profile.get("languages", [])]
    if preferred_langs:
        df_filtered = df_filtered[
            df_filtered["language"].str.lower().isin(preferred_langs)
        ]

    # Filter by "good first issue" label for beginners
    if difficulty == "beginner":
        beginner_mask = df_filtered["labels"].str.contains(
            "good first issue|beginner|easy|starter", case=False, na=False
        )
        df_beginner = df_filtered[beginner_mask]
        if len(df_beginner) >= top_n:
            df_filtered = df_beginner

    return df_filtered.sample(min(top_n, len(df_filtered)))

def smart_recommend(user_profile: dict, top_n: int = 10,
                     threshold: float = 0.1) -> pd.DataFrame:
    """
    Use TF-IDF if scores are above threshold, otherwise use fallback.
    """
    recs = recommend_issues(user_profile, top_n)

    if recs["similarity_score"].max() < threshold:
        df = pd.read_csv("model/issues_index.csv")
        return fallback_recommend(user_profile, df, top_n)

    return recs
```

## Phase 9 – API Integration with FastAPI

### 9.1 FastAPI Endpoint

```python
# api/main.py
from fastapi import FastAPI
from pydantic import BaseModel
from typing import List, Optional
from recommend import smart_recommend

app = FastAPI(title="OpenCollab ML API", version="1.0.0")

class UserProfile(BaseModel):
    languages: List[str] = ["Python"]
    difficulty: str = "beginner"
    topics: List[str] = []
    keywords: List[str] = []

class IssueRecommendation(BaseModel):
    issue_id: str
    repo_name: str
    issue_title: str
    language: str
    difficulty: str
    labels: str
    similarity_score: float

@app.post("/recommend", response_model=List[IssueRecommendation])
def get_recommendations(user: UserProfile, top_n: int = 10):
    """Return top-N recommended issues for a user profile."""
    results = smart_recommend(user.dict(), top_n=top_n)
    return results.to_dict(orient="records")

@app.get("/health")
def health_check():
    return {"status": "ok", "model": "tfidf-v1"}
```
